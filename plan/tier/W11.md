import { Agent } from '@strands-agents/sdk';
import type { AppConfig } from '@promptsheon/shared';

export function createModel(config: AppConfig) {
  // Factory pattern: create model based on config
  const provider = config.llm.defaultProvider;
  const modelId = config.llm.defaultModel;

  switch (provider) {
    case 'anthropic':
      return new AnthropicModel({ modelId });
    case 'bedrock':
      return new BedrockModel({ modelId });
    case 'openai':
    default:
      return new OpenAIModel({ modelId });
  }
}

### `packages/server/src/agents/invocation.ts`
```typescript
import { Agent } from '@strands-agents/sdk';
import { SlidingWindowConversationManager } from '@strands-agents/sdk';
import { DefaultModelRetryStrategy } from '@strands-agents/sdk';
import type { AppConfig, CapabilityVersion, Manifest, Execution } from '@promptsheon/shared';
import { createModel } from './model';
import { lookupCapabilityTool, recordExecutionTool } from './tools';

export class InvocationAgent {
  private agent: Agent;

  constructor(config: AppConfig) {
    this.agent = new Agent({
      model: createModel(config),
      systemPrompt: `You are a capability execution engine. When invoked:
1. Look up the capability manifest
2. Execute the prompt with the provided inputs
3. Record the execution result
4. Return the outputs`,
      tools: [lookupCapabilityTool, recordExecutionTool],
      conversationManager: new SlidingWindowConversationManager({
        windowSize: 20,
      }),
      retryStrategy: new DefaultModelRetryStrategy({
        maxAttempts: config.llm.maxRetries,
      }),
    });
  }

  async invoke(
    capabilityVersionId: string,
    inputs: Record<string, unknown>,
    options: { environment?: string; traceId?: string; onEvent?: (event: any) => void } = {},
  ): Promise<Execution> {
    const startTime = Date.now();

    const prompt = JSON.stringify({
      action: 'invoke',
      capabilityVersionId,
      inputs,
      environment: options.environment ?? '',
      traceId: options.traceId ?? '',
    });

    const result = await this.agent.invoke(prompt);

    // Parse execution from result
    const execution: Execution = {
      id: crypto.randomUUID(),
      capabilityVersionId,
      timestamp: new Date().toISOString(),
      inputs: JSON.stringify(inputs),
      outputs: JSON.stringify(result.message),
      model: '',
      provider: '',
      latencyMs: Date.now() - startTime,
      costUsd: 0,
      promptTokens: 0,
      completionTokens: 0,
      totalTokens: 0,
      error: '',
      traceId: options.traceId ?? '',
      environment: options.environment ?? '',
    };

    return execution;
  }

  async invokeStreaming(
    capabilityVersionId: string,
    inputs: Record<string, unknown>,
    onEvent: (event: any) => void,
  ): Promise<Execution> {
    const startTime = Date.now();
    const prompt = JSON.stringify({ action: 'invoke', capabilityVersionId, inputs });

    for await (const event of this.agent.stream(prompt)) {
      onEvent(event);
    }

    return {
      id: crypto.randomUUID(),
      capabilityVersionId,
      timestamp: new Date().toISOString(),
      inputs: JSON.stringify(inputs),
      outputs: '{}',
      model: '',
      provider: '',
      latencyMs: Date.now() - startTime,
      costUsd: 0,
      promptTokens: 0,
      completionTokens: 0,
      totalTokens: 0,
      error: '',
      traceId: '',
      environment: '',
    };
  }
}
```

## Acceptance Criteria

- [ ] Agent uses Strands `tool()` for DB operations
- [ ] Agent uses `SlidingWindowConversationManager` for context
- [ ] Agent uses `DefaultModelRetryStrategy` for retries
- [ ] Agent uses `Streaming` (async iterators) for real-time output
- [ ] Agent uses `StructuredOutput` for execution results
- [ ] Model factory supports OpenAI, Anthropic, Bedrock
- [ ] `invoke` returns `Execution` with correct timestamps
- [ ] `invokeStreaming` calls `onEvent` for each stream event
- [ ] Agent system prompt instructs correct behavior
- [ ] Tools are registered via Strands `tool()` with Zod schemas
- [ ] All DB operations go through repos (not raw SQL)
- [ ] No direct LLM HTTP calls (all via Strands SDK)
