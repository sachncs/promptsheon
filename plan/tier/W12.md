```typescript
import type { Capability, EvalRun } from '@promptsheon/shared';

export class PerformanceDetector {
  detect(capability: Capability, recentEvals: EvalRun[]): boolean {
    // If no evals, no detection
    if (recentEvals.length === 0) return false;

    // Check if latest eval score is below threshold
    const latest = recentEvals[0];
    return latest.score < capability.selfEvolveMinScore;
  }
}
```

```typescript
import type { CasStore } from '@promptsheon/shared/cas';
import type { Manifest } from '@promptsheon/shared';

export class CasPromptLoader {
  constructor(private cas: CasStore) {}

  async load(hash: string): Promise<Manifest> {
    const obj = await this.cas.readObject(hash);
    if (obj.type !== 'blob') throw new Error('expected blob');
    return JSON.parse(obj.data.toString());
  }

  async save(manifest: Manifest): Promise<string> {
    const data = Buffer.from(JSON.stringify(manifest));
    return this.cas.writeObject({ type: 'blob', data });
  }
}
```

```typescript
  failingCases: Array<{ inputs: unknown; expected: unknown; actual: unknown }>;
  evaluationSummary: string;
}

  revisedManifest: Manifest;
  changes: string[];
  reasoning: string;
}

export class LLMRevisionStrategy {
  private agent: Agent;

  constructor(config: AppConfig) {
    this.agent = new Agent({
      model: createModel(config),
      systemPrompt: `You are a prompt revision agent. Your job is to improve a prompt based on evaluation failures.

Given:
- The current manifest (system prompt, tools, parameters)
- Failing test cases with inputs, expected outputs, and actual outputs
- An evaluation summary

Revise the manifest to fix the failures while preserving correct behavior.

Output a JSON object with:
- revisedManifest: the improved manifest
- changes: list of changes made
- reasoning: explanation of why changes were made`,
    });
  }

  async revise(request: ReviseRequest): Promise<ReviseResponse> {
    const result = await this.agent.invoke(JSON.stringify(request));
    return JSON.parse(result.message);
  }
}
```

### `packages/server/src/agents/evolution/validator.ts`
```typescript
import type { Manifest, EvalRun } from '@promptsheon/shared';
import { Agent } from '@strands-agents/sdk';

export class HarnessValidator {
  private agent: Agent;

  constructor(config: AppConfig) {
    this.agent = new Agent({
      model: createModel(config),
      systemPrompt: `You are an evaluation judge. Score the quality of AI outputs against expected results.`,
    });
  }

  async validate(
    manifest: Manifest,
    testCases: Array<{ inputs: unknown; expected: unknown }>,
  ): Promise<EvalRun> {
    const results = [];
    for (const testCase of testCases) {
      const result = await this.agent.invoke(JSON.stringify({
        manifest,
        inputs: testCase.inputs,
      }));
      results.push({
        actual: result.message,
        expected: testCase.expected,
        passed: this.compareOutputs(result.message, testCase.expected),
      });
    }

    const passed = results.filter(r => r.passed).length;
    return {
      id: crypto.randomUUID(),
      releaseId: '',
      datasetId: '',
      scorer: 'llm-judge',
      score: passed / results.length,
      passed,
      failed: results.length - passed,
      total: results.length,
      status: passed === results.length ? 'passed' : 'failed',
      startedAt: new Date().toISOString(),
      finishedAt: new Date().toISOString(),
    };
  }

  private compareOutputs(actual: string, expected: unknown): boolean {
    // Semantic comparison using LLM
    return actual === JSON.stringify(expected);
  }
}
```

### `packages/server/src/agents/evolution/promoter.ts`
```typescript
import type { CasStore } from '@promptsheon/shared/cas';
import type { Manifest, CapabilityVersion } from '@promptsheon/shared';
import { createCommit, createTree, createBlob } from '@promptsheon/shared/cas/object';
import { commit as casCommit } from '@promptsheon/shared/cas/commit';

export class Promoter {
  constructor(
    private cas: CasStore,
    private versionRepo: VersionRepo,
    private releaseRepo: ReleaseRepo,
  ) {}

  async promote(
    capabilityId: string,
    revisedManifest: Manifest,
    targetEnv: string,
    author: string,
  ): Promise<CapabilityVersion> {
    // 1. Save manifest to CAS
    const manifestHash = await this.cas.writeObject({
      type: 'blob',
      data: Buffer.from(JSON.stringify(revisedManifest)),
    });

    // 2. Create tree
    const treeHash = await this.cas.writeObject({
      type: 'tree',
      entries: [{ name: 'manifest.json', hash: manifestHash, type: 'blob' }],
    });

    // 3. Create commit
    const commitHash = await casCommit(this.cas, {
      type: 'commit',
      treeHash,
      parents: [],
      author,
      message: `self-evolve: revise manifest for capability ${capabilityId}`,
      telemetry: { capabilityId, targetEnv },
    }, 'main');

    // 4. Create new version
    const latestVersion = await this.versionRepo.findLatest(capabilityId);
    const newVersion = latestVersion ? latestVersion.version + 1 : 1;

    const version = await this.versionRepo.create({
      capabilityId,
      version: newVersion,
      manifest: JSON.stringify(revisedManifest),
      manifestHash,
      createdBy: author,
    });

    return version;
  }
}
```

## Acceptance Criteria

- [ ] Self-evolution follows closed-loop: detect→revise→validate→promote
- [ ] `PerformanceDetector` checks eval score against threshold
- [ ] `LLMRevisionStrategy` uses Strands Agent for manifest revision
- [ ] `HarnessValidator` uses Strands Agent for LLM-as-judge evaluation
- [ ] `Promoter` writes to CAS + creates new version
- [ ] `EvolutionAgent` uses Strands `Graph` for orchestration
- [ ] `Snapshot` taken before revision (for rollback)
- [ ] `Hooks` used for lifecycle events (BeforeToolCall, AfterModelCall)
- [ ] `ConversationManager` manages context for long revision conversations
- [ ] `Prompts` (Agent SOPs) define the evolution workflow
- [ ] State transitions: idle→detected→revising→validating→promoted/rejected
- [ ] Cooldown period respected between cycles
- [ ] Max concurrent cycles enforced
