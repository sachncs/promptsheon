### `packages/web/src/lib/api.ts`
API client:
```typescript
import axios from 'axios';
import type {
  Workspace, Project, Capability, CapabilityVersion,
  Release, Execution, Dataset, EvalRun, AlertRule, Alert,
} from '@promptsheon/shared';

const client = axios.create({
  baseURL: '/api',
  headers: { 'Content-Type': 'application/json' },
});

// Workspace
export const workspaceApi = {
  list: () => client.get<{ items: Workspace[]; total: number }>('/workspaces'),
  get: (id: string) => client.get<Workspace>(`/workspaces/${id}`),
  create: (data: any) => client.post<Workspace>('/workspaces', data),
  update: (id: string, data: any) => client.put<Workspace>(`/workspaces/${id}`, data),
  delete: (id: string) => client.delete(`/workspaces/${id}`),
};

// Project
export const projectApi = {
  list: (workspaceId: string) => client.get(`/projects?workspaceId=${workspaceId}`),
  get: (id: string) => client.get(`/projects/${id}`),
  create: (data: any) => client.post('/projects', data),
  update: (id: string, data: any) => client.put(`/projects/${id}`, data),
  delete: (id: string) => client.delete(`/projects/${id}`),
};

// ... similar for all entities

// SSE
export function subscribeSSE(channel: string, onEvent: (event: any) => void): () => void {
  const eventSource = new EventSource(`/api/events/${channel}`);
  eventSource.onmessage = (e) => onEvent(JSON.parse(e.data));
  eventSource.onerror = () => {
    // Reconnect after 3s
    setTimeout(() => {
      eventSource.close();
      subscribeSSE(channel, onEvent);
    }, 3000);
  };
  return () => eventSource.close();
}
```
