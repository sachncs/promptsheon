## Files to Create

### `packages/shared/src/audit/chain.ts`
```typescript
import { createHash } from 'crypto';
import type Database from 'better-sqlite3';
import type { AuditEntry } from '../types';

export class AuditChain {
  constructor(private db: Database.Database) {}

  async append(entry: Omit<AuditEntry, 'id' | 'previousHash' | 'entryHash' | 'timestampStr'>): Promise<AuditEntry> {
    // Get last hash from chain state
    const chainState = this.db.prepare('SELECT last_hash FROM audit_chain_state WHERE id = 0').get() as any;
    const previousHash = chainState?.last_hash ?? '';

    // Compute entry hash
    const hashInput = JSON.stringify({ ...entry, previousHash });
    const entryHash = createHash('sha256').update(hashInput).digest('hex');

    // Generate ID
    const id = crypto.randomUUID();
    const now = new Date().toISOString();

    // Insert entry
    this.db.prepare(`
      INSERT INTO audit_entries (id, user_id, action, resource, details, timestamp, previous_hash, entry_hash, timestamp_str, resource_kind, resource_id)
      VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
    `).run(
      id, entry.userId, entry.action, entry.resource, entry.details,
      now, previousHash, entryHash, now, entry.resourceKind, entry.resourceId,
    );

    // Update chain state
    this.db.prepare('UPDATE audit_chain_state SET last_hash = ?, last_rowid = (SELECT last_insert_rowid()) WHERE id = 0')
      .run(entryHash);

    return {
      id, userId: entry.userId, action: entry.action, resource: entry.resource,
      details: entry.details, timestamp: now, previousHash, entryHash,
      timestampStr: now, resourceKind: entry.resourceKind, resourceId: entry.resourceId,
    };
  }

  verify(): { valid: boolean; brokenAt?: string } {
    const entries = this.db.prepare('SELECT * FROM audit_entries ORDER BY rowid ASC').all() as AuditEntry[];
    let expectedPrevious = '';

    for (const entry of entries) {
      if (entry.previousHash !== expectedPrevious) {
        return { valid: false, brokenAt: entry.id };
      }
      const hashInput = JSON.stringify({
        userId: entry.userId, action: entry.action, resource: entry.resource,
        details: entry.details, previousHash: entry.previousHash,
      });
      const computedHash = createHash('sha256').update(hashInput).digest('hex');
      if (computedHash !== entry.entryHash) {
        return { valid: false, brokenAt: entry.id };
      }
      expectedPrevious = entry.entryHash;
    }

    return { valid: true };
  }

  getChainState(): { lastHash: string; lastRowid: number } {
    return this.db.prepare('SELECT last_hash as lastHash, last_rowid as lastRowid FROM audit_chain_state WHERE id = 0')
      .get() as any;
  }
}
```

## Acceptance Criteria

- [ ] Each entry's `previousHash` matches the previous entry's `entryHash`
- [ ] Entry hash is SHA-256 of (entry fields + previousHash)
- [ ] Chain state singleton row is maintained
- [ ] `verify` walks entire chain and detects breaks
- [ ] `verify` returns first broken entry ID
- [ ] Append is atomic (insert + update in transaction)
- [ ] Hash computation matches Go implementation
- [ ] Append-only (UPDATE trigger prevents modification)
- [ ] Works with empty chain (first entry has empty previousHash)
