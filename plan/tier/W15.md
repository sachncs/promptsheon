## Files to Create

Each route file follows this pattern:
```typescript
import type { FastifyInstance } from 'fastify';

export async function registerXxxRoutes(app: FastifyInstance): Promise<void> {
  const repo = app.xxxRepo;

  // List
  app.get('/api/xxx', async (request, reply) => {
    const { page, pageSize } = PaginationSchema.parse(request.query);
    const result = await repo.findMany({ page, pageSize });
    return reply.send(result);
  });

  // Get by ID
  app.get('/api/xxx/:id', async (request, reply) => {
    const { id } = request.params as { id: string };
    const item = await repo.findById(id);
    if (!item) return reply.code(404).send({ error: { code: 'NOT_FOUND', message: 'Not found' } });
    return reply.send(item);
  });

  // Create
  app.post('/api/xxx', async (request, reply) => {
    const data = CreateXxxSchema.parse(request.body);
    const item = await repo.create(data);
    return reply.code(201).send(item);
  });

  // Update
  app.put('/api/xxx/:id', async (request, reply) => {
    const { id } = request.params as { id: string };
    const data = UpdateXxxSchema.parse(request.body);
    const item = await repo.update(id, data);
    return reply.send(item);
  });

  // Delete
  app.delete('/api/xxx/:id', async (request, reply) => {
    const { id } = request.params as { id: string };
    await repo.delete(id);
    return reply.code(204).send();
  });
}
```

### Special Endpoints

**`/api/invoke`** — POST, invokes capability via Strands agent
```typescript
app.post('/api/invoke', async (request, reply) => {
  const { capabilityVersionId, inputs, environment, traceId } = InvokeExecutionSchema.parse(request.body);
  const execution = await invocationAgent.invoke(capabilityVersionId, inputs, { environment, traceId });
  return reply.send(execution);
});
```

**`/api/self-evolve/run`** — POST, triggers self-evolution cycle
```typescript
app.post('/api/self-evolve/run', async (request, reply) => {
  const { capabilityId } = request.body as { capabilityId: string };
  const state = await evolutionAgent.runCycle(capabilityId);
  return reply.send(state);
});
```

**`/api/eval/run`** — POST, runs evaluation
```typescript
app.post('/api/eval/run', async (request, reply) => {
  const evalRun = CreateEvalRunSchema.parse(request.body);
  const result = await evaluationAgent.runEval(evalRun, cases, getActual);
  return reply.send(result);
});
```

**`/api/events/:channel`** — GET, SSE streaming
```typescript
app.get('/api/events/:channel', async (request, reply) => {
  const { channel } = request.params as { channel: string };
  const client = new SseServerClient(reply, crypto.randomUUID());
  sseHub.subscribe(channel, client);
  request.raw.on('close', () => sseHub.unsubscribe(channel, client));
});
```

## Acceptance Criteria

- [ ] All 100+ Go API endpoints have TypeScript equivalents
- [ ] All routes use Zod validation for request bodies
- [ ] All routes use pagination schema for list endpoints
- [ ] All routes return correct HTTP status codes
- [ ] All routes use error handler from W06
- [ ] SSE endpoint sets correct headers
- [ ] Invoke endpoint uses Strands agent
- [ ] Self-evolve endpoint triggers evolution cycle
- [ ] Eval endpoint runs evaluation
- [ ] All routes are registered in index.ts
- [ ] No business logic in route handlers (delegate to repos/agents)
- [ ] Route params are typed (not `any`)
- [ ] Query params are validated
- [ ] Request body is validated
- [ ] Response is typed
