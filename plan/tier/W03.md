## Files to Create

### `packages/server/src/config/env.ts`
Reads `PROMPTSHEON_*` env vars with defaults:
```typescript
export function loadConfig(): AppConfig {
  return {
    server: {
      port: parseInt(process.env.PROMPTSHEON_PORT || '8080', 10),
      host: process.env.PROMPTSHEON_HOST || '127.0.0.1',
      dbPath: process.env.PROMPTSHEON_DB_PATH || './promptsheon.db',
      casPath: process.env.PROMPTSHEON_CAS_PATH || './.promptsheon',
      frontendPath: process.env.PROMPTSHEON_FRONTEND_PATH || './frontend/dist',
      corsOrigin: process.env.PROMPTSHEON_CORS_ORIGIN || '*',
      logLevel: process.env.PROMPTSHEON_LOG_LEVEL || 'info',
    },
    llm: {
      defaultProvider: process.env.PROMPTSHEON_LLM_PROVIDER || 'openai',
      defaultModel: process.env.PROMPTSHEON_LLM_MODEL || 'gpt-4',
      apiKeyEnvVar: process.env.PROMPTSHEON_LLM_API_KEY_ENV || 'OPENAI_API_KEY',
      maxRetries: parseInt(process.env.PROMPTSHEON_LLM_MAX_RETRIES || '5', 10),
      timeoutMs: parseInt(process.env.PROMPTSHEON_LLM_TIMEOUT_MS || '120000', 10),
    },
    auth: {
      enabled: process.env.PROMPTSHEON_AUTH_ENABLED !== 'false',
      jwtSecret: process.env.PROMPTSHEON_JWT_SECRET || '',
    },
    selfEvolve: {
      enabled: process.env.PROMPTSHEON_SELF_EVOLVE_ENABLED === 'true',
      defaultCooldownSec: parseInt(process.env.PROMPTSHEON_SELF_EVOLVE_COOLDOWN_SEC || '900', 10),
      maxConcurrentCycles: parseInt(process.env.PROMPTSHEON_SELF_EVOLVE_MAX_CONCURRENT || '3', 10),
    },
  };
}
```

### `packages/server/src/config/validate.ts`
Validates loaded config:
```typescript
export function validateConfig(config: AppConfig): void {
  if (config.auth.enabled && !config.auth.jwtSecret) {
    throw new Error('PROMPTSHEON_JWT_SECRET is required when auth is enabled');
  }
  if (config.server.port < 1 || config.server.port > 65535) {
    throw new Error('PROMPTSHEON_PORT must be between 1 and 65535');
  }
}
```
