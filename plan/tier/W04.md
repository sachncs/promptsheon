## Files to Create

### `packages/server/src/db/connection.ts`
```typescript
import Database from 'better-sqlite3';
import type { AppConfig } from '@promptsheon/shared';

export function createConnection(config: AppConfig): Database.Database {
  const db = new Database(config.server.dbPath);
  db.pragma('journal_mode = WAL');
  db.pragma('foreign_keys = ON');
  db.pragma('busy_timeout = 5000');
  return db;
}
```

### `packages/server/src/db/migrate.ts`
```typescript
import { readdir, readFile } from 'fs/promises';
import { join } from 'path';

export interface MigrationRecord {
  version: number;
  name: string;
  appliedAt: string;
}

export async function runMigrations(db: Database.Database): Promise<void> {
  // 1. Create _migrations table if not exists
  db.exec(`
    CREATE TABLE IF NOT EXISTS _migrations (
      version INTEGER PRIMARY KEY,
      name TEXT NOT NULL,
      applied_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
    )
  `);

  // 2. Read applied migrations
  const applied = new Set(
    db.prepare('SELECT version FROM _migrations').all().map((r: any) => r.version)
  );

  // 3. Read migration files from disk
  const migrationDir = join(__dirname, '../../db/migrations');
  const files = await readdir(migrationDir);
  const ups = files
    .filter(f => f.endsWith('.up.sql'))
    .map(f => ({
      version: parseInt(f.split('_')[0], 10),
      name: f,
      path: join(migrationDir, f),
    }))
    .sort((a, b) => a.version - b.version);

  // 4. Apply pending migrations in transaction
  for (const migration of ups) {
    if (applied.has(migration.version)) continue;
    const sql = await readFile(migration.path, 'utf-8');
    db.transaction(() => {
      db.exec(sql);
      db.prepare('INSERT INTO _migrations (version, name) VALUES (?, ?)')
        .run(migration.version, migration.name);
    })();
    console.log(`Applied migration ${migration.name}`);
  }
}
```

### `packages/server/src/db/transaction.ts`
```typescript
import type Database from 'better-sqlite3';

export function inTransaction<T>(db: Database.Database, fn: () => T): T {
  return db.transaction(fn)();
}
```

## Acceptance Criteria

- [ ] SQLite WAL mode enabled
- [ ] Foreign keys enforced
- [ ] Busy timeout set to 5s
- [ ] Migrations run in version order
- [ ] Already-applied migrations are skipped
- [ ] Each migration runs in a transaction
- [ ] `_migrations` table tracks applied versions
- [ ] `inTransaction` helper wraps callbacks
- [ ] Connection is closeable (returns db with `.close()`)
- [ ] Works with in-memory SQLite (`:memory:`) for tests
