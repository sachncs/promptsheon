### `packages/server/src/middleware/auth.ts`
```typescript
import type { FastifyRequest, FastifyReply } from 'fastify';
import { createHash } from 'crypto';
import type { AppConfig } from '@promptsheon/shared';
import type { ApiKeyRepo } from '../repos/api-key';

export function authMiddleware(config: AppConfig, apiKeyRepo: ApiKeyRepo) {
  return async (request: FastifyRequest, reply: FastifyReply) => {
    if (!config.auth.enabled) {
      // Inject system user
      (request as any).userId = 'api';
      return;
    }

    const authHeader = request.headers.authorization;
    if (!authHeader) {
      return reply.code(401).send({ error: { code: 'UNAUTHORIZED', message: 'Missing authorization header' } });
    }

    if (authHeader.startsWith('Bearer ')) {
      const token = authHeader.slice(7);
      const keyHash = createHash('sha256').update(token).digest('hex');
      const apiKey = await apiKeyRepo.findByKeyHash(keyHash);

      if (!apiKey || apiKey.revoked) {
        return reply.code(401).send({ error: { code: 'UNAUTHORIZED', message: 'Invalid API key' } });
      }

      if (apiKey.expiresAt && new Date(apiKey.expiresAt) < new Date()) {
        return reply.code(401).send({ error: { code: 'UNAUTHORIZED', message: 'API key expired' } });
      }

      (request as any).userId = apiKey.userId;
      (request as any).userRole = apiKey.role;
      return;
    }

    return reply.code(401).send({ error: { code: 'UNAUTHORIZED', message: 'Invalid authorization format' } });
  };
}
```

### `packages/server/src/middleware/ssrf.ts`
```typescript
import { URL } from 'url';

const BLOCKED_HOSTS = ['localhost', '127.0.0.1', '0.0.0.0', '::1', '169.254.169.254'];
const BLOCKED_SCHEMES = ['file:', 'ftp:', 'gopher:'];

export function validateUrl(urlString: string): { valid: boolean; error?: string } {
  try {
    const url = new URL(urlString);

    if (BLOCKED_SCHEMES.includes(url.protocol)) {
      return { valid: false, error: `Blocked scheme: ${url.protocol}` };
    }

    if (BLOCKED_HOSTS.includes(url.hostname)) {
      return { valid: false, error: `Blocked host: ${url.hostname}` };
    }

    // Block private IPs
    const ip = url.hostname;
    if (/^(10\.|172\.(1[6-9]|2\d|3[01])\.|192\.168\.)/.test(ip)) {
      return { valid: false, error: `Blocked private IP: ${ip}` };
    }

    return { valid: true };
  } catch {
    return { valid: false, error: 'Invalid URL' };
  }
}

export function ssrfProtection() {
  return async (request: FastifyRequest, reply: FastifyReply) => {
    const body = request.body as any;
    if (body?.url) {
      const validation = validateUrl(body.url);
      if (!validation.valid) {
        return reply.code(400).send({
          error: { code: 'BAD_REQUEST', message: validation.error },
        });
      }
    }
  };
}
```

### `packages/server/src/middleware/idempotency.ts`
```typescript
import type { FastifyRequest, FastifyReply } from 'fastify';
import { createHash } from 'crypto';
import type { IdempotencyRepo } from '../repos/idempotency';

export function idempotencyMiddleware(idempotencyRepo: IdempotencyRepo) {
  return async (request: FastifyRequest, reply: FastifyReply) => {
    const idempotencyKey = request.headers['idempotency-key'] as string;
    if (!idempotencyKey) return;

    const key = createHash('sha256')
      .update(`${request.method}:${request.url}:${idempotencyKey}`)
      .digest('hex');

    const cached = await idempotencyRepo.get(key);
    if (cached) {
      return reply
        .code(cached.statusCode)
        .headers(JSON.parse(cached.headers))
        .send(cached.body);
    }

    // Store response after handler
    reply.raw.on('finish', async () => {
      await idempotencyRepo.set(key, {
        statusCode: reply.statusCode,
        headers: JSON.stringify(reply.getHeaders()),
        body: '',
        expiresAt: new Date(Date.now() + 86400000).toISOString(), // 24h
      });
    });
  };
}
```

### `packages/server/src/middleware/errors.ts`
```typescript
import type { FastifyError } from 'fastify';
import { AppError, ErrorCode, errorToStatus, errorToResponse } from '@promptsheon/shared';

export function errorHandler(error: FastifyError, request: FastifyRequest, reply: FastifyReply) {
  if (error instanceof AppError) {
    return reply.code(errorToStatus(error)).send(errorToResponse(error));
  }

  // Zod validation error
  if (error.message.includes('Validation')) {
    return reply.code(422).send({
      error: { code: 'VALIDATION_ERROR', message: error.message },
    });
  }

  // Default
  return reply.code(500).send({
    error: { code: 'INTERNAL_ERROR', message: 'Internal server error' },
  });
}
```

### `packages/server/src/middleware/audit.ts`
```typescript
import type { FastifyRequest, FastifyReply } from 'fastify';
import type { AuditChain } from '@promptsheon/shared/audit';

export function auditMiddleware(auditChain: AuditChain) {
  return async (request: FastifyRequest, reply: FastifyReply) => {
    const startTime = Date.now();

    reply.raw.on('finish', async () => {
      if (request.method === 'GET') return; // Skip reads

      await auditChain.append({
        userId: (request as any).userId ?? 'anonymous',
        action: `${request.method} ${request.url}`,
        resource: request.url.split('/')[2] ?? 'unknown',
        details: JSON.stringify({
          statusCode: reply.statusCode,
          latencyMs: Date.now() - startTime,
        }),
        timestamp: new Date().toISOString(),
        resourceKind: request.url.split('/')[2] ?? 'unknown',
        resourceId: (request.params as any)?.id ?? '',
      });
    });
  };
}
```

## Acceptance Criteria

- [ ] Auth middleware validates Bearer tokens via API key hash
- [ ] Auth middleware injects userId into request
- [ ] Auth middleware can be disabled via config
- [ ] CORS config uses configured origin
- [ ] Rate limiting uses userId or IP as key
- [ ] SSRF protection blocks localhost, private IPs, metadata endpoints
- [ ] Idempotency middleware caches responses by key
- [ ] Idempotency middleware returns cached response on replay
- [ ] Error handler converts AppError to HTTP response
- [ ] Error handler converts Zod errors to 422
- [ ] Audit middleware logs non-GET requests
- [ ] Audit middleware uses hash chain from W10
- [ ] All middleware is composable via Fastify hooks
- [ ] No middleware depends on route handlers
