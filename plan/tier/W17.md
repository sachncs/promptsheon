### `packages/server/src/scheduler/scheduler.ts`
```typescript
import type { ScheduleRepo } from '../repos/schedule';
import type { SseHub } from '../sse/hub';

export class Scheduler {
  private interval: NodeJS.Timeout | null = null;
  private running = false;

  constructor(
    private scheduleRepo: ScheduleRepo,
    private sseHub: SseHub,
    private handlers: Map<string, (schedule: Schedule) => Promise<void>>,
  ) {}

  start(pollIntervalMs = 10000): void {
    this.interval = setInterval(() => this.poll(), pollIntervalMs);
    this.poll(); // Initial poll
  }

  stop(): void {
    if (this.interval) clearInterval(this.interval);
  }

  async poll(): Promise<void> {
    if (this.running) return;
    this.running = true;

    try {
      const now = new Date().toISOString();
      const dueSchedules = await this.scheduleRepo.findDue(now);

      for (const schedule of dueSchedules) {
        const handler = this.handlers.get(schedule.kind);
        if (handler) {
          try {
            await handler(schedule);
            await this.scheduleRepo.markFired(schedule.id, now);
            this.sseHub.broadcast({
              type: 'status',
              data: { scheduleId: schedule.id, status: 'fired' },
              timestamp: new Date().toISOString(),
            });
          } catch (e) {
            console.error(`Schedule ${schedule.id} failed:`, e);
            this.sseHub.broadcast({
              type: 'error',
              data: { scheduleId: schedule.id, error: String(e) },
              timestamp: new Date().toISOString(),
            });
          }
        }
      }
    } finally {
      this.running = false;
    }
  }

  registerHandler(kind: string, handler: (schedule: Schedule) => Promise<void>): void {
    this.handlers.set(kind, handler);
  }
}
```

### `packages/server/src/scheduler/webhook-delivery.ts`
```typescript
import type { WebhookEndpoint } from '@promptsheon/shared';
import type { WebhookRepo } from '../repos/webhook';
import { createHmac } from 'crypto';

export class WebhookDelivery {
  constructor(private webhookRepo: WebhookRepo) {}

  async deliver(event: string, payload: unknown): Promise<void> {
    const endpoints = await this.webhookRepo.findByEvent(event);

    for (const endpoint of endpoints) {
      if (!endpoint.active) continue;

      try {
        const body = JSON.stringify({ event, payload, timestamp: new Date().toISOString() });

        const headers: Record<string, string> = {
          'Content-Type': 'application/json',
          'X-Promptsheon-Event': event,
        };

        if (endpoint.secretCiphertext) {
          const signature = createHmac('sha256', endpoint.secretCiphertext.toString())
            .update(body)
            .digest('hex');
          headers['X-Promptsheon-Signature'] = signature;
        }

        await fetch(endpoint.url, {
          method: 'POST',
          headers,
          body,
          signal: AbortSignal.timeout(30000), // 30s timeout
        });
      } catch (e) {
        console.error(`Webhook delivery to ${endpoint.url} failed:`, e);
      }
    }
  }
}
```

## Acceptance Criteria

- [ ] Scheduler polls due schedules every 10 seconds
- [ ] Scheduler processes each schedule exactly once per poll
- [ ] Scheduler broadcasts status via SSE
- [ ] Alert checker supports 4 alert types
- [ ] Alert checker queries recent data within time window
- [ ] Alert checker creates alerts when thresholds exceeded
- [ ] Webhook delivery sends POST with JSON payload
- [ ] Webhook delivery signs payloads with HMAC
- [ ] Webhook delivery has 30s timeout
- [ ] Webhook delivery is best-effort (doesn't block scheduler)
- [ ] All handlers are registered via `registerHandler`
- [ ] No handler depends on route handlers or agents
