## Files to Create

### `packages/server/src/settings/resolver.ts`
Three-layer precedence: default < env < DB

```typescript
import type { AppConfig } from '@promptsheon/shared';
import type { SystemConfigRepo } from '../repos/system-config';

export class SettingsResolver {
  private cache = new Map<string, unknown>();
  private cacheExpiry = new Map<string, number>();
  private readonly CACHE_TTL_MS = 30_000; // 30s

  constructor(
    private defaults: Record<string, unknown>,
    private envSource: Record<string, string>,
    private dbRepo: SystemConfigRepo,
  ) {}

  async get<T = unknown>(key: string): Promise<T> {
    // Check cache
    const cached = this.cache.get(key);
    const expiry = this.cacheExpiry.get(key);
    if (cached !== undefined && expiry && Date.now() < expiry) {
      return cached as T;
    }

    // Layer 1: DB
    const dbValue = await this.dbRepo.get(key);
    if (dbValue && !dbValue.tombstone) {
      const parsed = JSON.parse(dbValue.value);
      this.setCache(key, parsed);
      return parsed as T;
    }

    // Layer 2: Env
    const envKey = key.replace(/\./g, '_').toUpperCase();
    const envValue = this.envSource[envKey] ?? this.envSource[`PROMPTSHEON_${envKey}`];
    if (envValue !== undefined) {
      const parsed = JSON.parse(envValue);
      this.setCache(key, parsed);
      return parsed as T;
    }

    // Layer 3: Default
    const defaultValue = this.defaults[key];
    if (defaultValue !== undefined) {
      return defaultValue as T;
    }

    return undefined as T;
  }

  async set(key: string, value: unknown, updatedBy?: string): Promise<void> {
    await this.dbRepo.set(key, JSON.stringify(value), updatedBy);
    this.invalidateCache(key);
    this.notifier?.notify(key);
  }

  async delete(key: string, updatedBy?: string): Promise<void> {
    await this.dbRepo.set(key, 'null', updatedBy, true); // tombstone
    this.invalidateCache(key);
    this.notifier?.notify(key);
  }

  async list(prefix?: string): Promise<Array<{ key: string; value: unknown }>> {
    const configs = await this.dbRepo.list(prefix);
    return configs
      .filter(c => !c.tombstone)
      .map(c => ({ key: c.key, value: JSON.parse(c.value) }));
  }

  async merge(updates: Record<string, unknown>, updatedBy?: string): Promise<void> {
    for (const [key, value] of Object.entries(updates)) {
      await this.set(key, value, updatedBy);
    }
  }

  private setCache(key: string, value: unknown): void {
    this.cache.set(key, value);
    this.cacheExpiry.set(key, Date.now() + this.CACHE_TTL_MS);
  }

  private invalidateCache(key: string): void {
    this.cache.delete(key);
    this.cacheExpiry.delete(key);
  }

  // For Strands MemoryManager integration
  private notifier?: SettingsNotifier;
  setNotifier(notifier: SettingsNotifier): void {
    this.notifier = notifier;
  }
}
```

### `packages/server/src/settings/notifier.ts`
```typescript
type Callback = (key: string) => void;

export class SettingsNotifier {
  private listeners = new Set<Callback>();

  subscribe(callback: Callback): () => void {
    this.listeners.add(callback);
    return () => this.listeners.delete(callback);
  }

  notify(key: string): void {
    for (const cb of this.listeners) {
      try {
        cb(key);
      } catch (e) {
        console.error('SettingsNotifier callback error:', e);
      }
    }
  }
}
```

### `packages/server/src/settings/secret-keys.ts`
```typescript
const SECRET_SEGMENTS = ['key', 'secret', 'password', 'token', 'credential'];

export function isSecretKey(key: string): boolean {
  const lower = key.toLowerCase();
  return SECRET_SEGMENTS.some(s => lower.includes(s));
}

export function maskSecret(value: string): string {
  if (value.length <= 8) return '****';
  return value.slice(0, 4) + '****' + value.slice(-4);
}
```

## Acceptance Criteria

- [ ] DB value takes precedence over env
- [ ] Env value takes precedence over default
- [ ] Missing value returns undefined (not error)
- [ ] Cache TTL is 30 seconds
- [ ] `set` invalidates cache immediately
- [ ] `delete` sets tombstone in DB
- [ ] `list` filters out tombstones
- [ ] `merge` sets multiple keys atomically
- [ ] Notifier fires on every set/delete
- [ ] Secret keys are masked in logs
- [ ] Thread-safe (single-threaded Node.js)
